```python
# --------------------------------------------------------------------------
# 失败现场
# --------------------------------------------------------------------------


@pytest.hookimpl(hookwrapper=True)
def pytest_runtest_makereport(item, call):
    """用例失败时把截图直接嵌进 HTML 报告。

    截图/录像/Trace 的落盘由运行参数控制（见 pyproject.toml 的 addopts），
    这里额外做的是"让人不用点开文件夹就能看到现场"。
    """
    outcome = yield
    report = outcome.get_result()

    if extras is None or report.when != "call" or not report.failed:
        return

    page = item.funcargs.get("page")
    if page is None or page.is_closed():
        return

    try:
        png = page.screenshot(full_page=True)
    except Exception:  # 页面已崩溃时截图会失败，不能因此掩盖真正的报错
        return

    report.extras = list(getattr(report, "extras", []))
    report.extras.append(
        extras.image(base64.b64encode(png).decode("utf-8"), name="失败截图")
    )
```

### 5.3 页面基类 —— `pages/base_page.py`

这是 POM 分层的核心，也是本次修复缺陷的主战场：

```python
"""页面基类：所有 Page Object 共用的基础能力。

设计原则：只封装"每个页面都会用到"的动作。
不要把 Playwright 的 API 再包一遍（比如给 click 加五层参数），
那会让 POM 变成负担，而不是资产。
"""

from __future__ import annotations

import re

from playwright.sync_api import Locator, Page, expect


class BasePage:
    """所有页面的父类。"""

    #: 页面路径，相对于 BASE_URL。子类按需覆盖。
    path: str = "/"

    def __init__(self, page: Page) -> None:
        self.page = page

    # ---------- 导航 ----------

    def open(self) -> "BasePage":
        """打开当前页面，返回自身以支持链式调用。"""
        self.page.goto(self.path)
        return self

    @property
    def url(self) -> str:
        return self.page.url

    # ---------- 定位 ----------

    def locator(self, selector: str, **kwargs) -> Locator:
        return self.page.locator(selector, **kwargs)

    def by_test_id(self, test_id: str) -> Locator:
        """按 data-test 属性定位。

        saucedemo 给关键元素都加了 data-test，用它定位比 CSS 类名稳定得多。
        建议团队内统一：data-test > 语义化 role > CSS 类名 > XPath。
        """
        return self.page.locator(f'[data-test="{test_id}"]')

    # ---------- 常用动作 ----------
    # Playwright 自带自动等待，这里不需要手写 sleep 或显式等待

    def click(self, selector: str) -> None:
        self.page.locator(selector).click()

    def fill(self, selector: str, value: str) -> None:
        self.page.locator(selector).fill(value)

    def text_of(self, selector: str) -> str:
        return self.page.locator(selector).inner_text().strip()

    def is_visible(self, selector: str) -> bool:
        """立即判断元素是否可见，不做任何等待。

        只能用在"页面已经稳定"的场景。凡是刚触发过跳转/异步刷新的地方，
        都用 wait_visible()，否则会拿到跳转前的旧状态 —— 这是 flaky 的头号来源。
        """
        return self.page.locator(selector).is_visible()

    def wait_visible(self, selector: str, timeout: float = 5000) -> bool:
        """等待元素出现并可见，超时返回 False（不抛异常）。

        适合"跳转之后判断到底成没成功"这类场景。

        注意：expect 是严格模式，selector 必须**唯一命中**一个元素。
        要判断"一类元素中至少有一个在"（比如商品列表），请显式加 .first。
        """
        try:
            expect(self.page.locator(selector)).to_be_visible(timeout=timeout)
        except AssertionError:
            return False
        return True

    # ---------- 断言辅助 ----------
    # 用 Playwright 的 expect（自动重试），而不是 assert + 立即取值，
    # 否则页面还没渲染完就会误报失败

    def expect_visible(self, selector: str, timeout: float | None = None) -> None:
        expect(self.page.locator(selector)).to_be_visible(timeout=timeout)

    def expect_text(self, selector: str, expected: str) -> None:
        expect(self.page.locator(selector)).to_have_text(expected)

    def expect_url_contains(self, fragment: str) -> None:
        """断言当前 URL 包含指定片段。

        注意必须传正则：to_have_url 收到字符串时是**精确匹配**，
        写 "./cart.html." 这种"模糊"字符串是匹配不上的。
        """
        expect(self.page).to_have_url(re.compile(re.escape(fragment)))
```

## 七、过程中修复的 3 个缺陷

第一轮跑出 3 个失败，**全部是我写的代码有问题，不是你的环境问题**。记录在此，因为它们恰好是 Web 自动化里最高频的三类翻车点。

### 缺陷 1：`to_have_url` 传字符串是精确匹配

**错误写法：**

```python
expect(self.page).to_have_url(f".*{fragment}.*")
```

**实际报错：**

```
AssertionError: Page URL expected to be 'https://www.saucedemo.com/.*/cart.html.*'
Actual value: https://www.saucedemo.com/cart.html
```

**原因**：`to_have_url` 收到字符串时做**精确比对**，通配符被当作普通字符。要模糊匹配必须传正则对象。

**修复：**

```python
expect(self.page).to_have_url(re.compile(re.escape(fragment)))
```

**影响范围**：`test_standard_user_login_success`、`test_cart_shows_added_item` 两条用例。

---

### 缺陷 2：`is_visible()` 是立即求值，不会等待

**错误写法：**

```python
def is_success(self) -> bool:
    return self.is_visible(self.COMPLETE_HEADER)
```

**现象**：点击"下单"后页面正在跳转，此时立即查询"成功页在不在"，得到的必然是 `False`。

**原因**：Playwright 有两套语义 —— `locator.click()` 这类**动作**自带自动等待，但 `locator.is_visible()` 这类**读取**是立即返回当前快照，不重试。

**修复**：新增 `wait_visible()`，内部用会重试的 `expect()`：

```python
def wait_visible(self, selector: str, timeout: float = 5000) -> bool:
    try:
        expect(self.page.locator(selector)).to_be_visible(timeout=timeout)
    except AssertionError:
        return False
    return True

def is_success(self) -> bool:
    # 下单后页面要跳转，必须等成功页出现才能判断，不能用 is_visible 立即求值
    return self.wait_visible(self.COMPLETE_HEADER)
```

**影响范围**：`test_full_checkout_flow`。

---

### 缺陷 3：Playwright 严格模式 —— 定位符必须唯一命中

**错误写法：**

```python
assert inventory.wait_visible(inventory.ITEM_CARD)   # ITEM_CARD = ".inventory_item"
```

**现象**：报错"登录后没有看到商品列表"，但从失败截图看商品列表**明明渲染正常**。

**原因**：`.inventory_item` 在页面上命中 **6 个**元素（每件商品一个）。Playwright 的 `expect()` 处于**严格模式**，定位符一次命中多个元素会被判定为失败 —— 它不会理解成"只要有一个可见就算过"。

**修复**：显式用 `.first` 表达"列表里有商品就行"这个真实意图：

```python
def expect_loaded(self) -> None:
    expect(self.page.locator(self.ITEM_CARD).first).to_be_visible()
```

**影响范围**：`test_standard_user_login_success`。

---

### 额外堵住的一个隐患（尚未爆发）

`cart_count()` 在点击加购后**立刻**读取角标，属于"读得太早"。这一轮侥幸通过，但在 CI 较慢的机器上迟早会飘。修复方式是让 `add_to_cart()` 等到按钮从 `Add to cart` 翻转成 `Remove` 再返回 —— 这既代表加入成功，也保证角标已刷新：

```python
    expect(card.get_by_role("button", name="Remove")).to_be_visible()
    return self
```

### 迁移步骤

**第 1 步 · 复制源码（主动跳过 `.venv`）**

> ⚠️ uv 输出了一条提示：`Failed to hardlink files; falling back to full copy.`
> 原因是**包缓存在 C 盘、项目在 D 盘，跨文件系统无法建立硬链接**，只能完整复制。
> 这只是性能提示，**不影响功能**，无需处理。若想消除提示，可设置环境变量 `UV_LINK_MODE=copy`。

### 一处观察（已复跑验证）

迁移后首次运行耗时 **55.08 秒**，迁移前为 **25.42 秒**，一度看起来像是迁移拖慢了速度。

**复跑一次为 30.52 秒**，确认差异来自首次运行的冷启动开销（浏览器在新路径下重建用户数据目录、磁盘缓存为空），**而非迁移本身导致，测试速度实际未受影响**。

### 另一个观察：外部站点的网络抖动（值得注意）

删除旧副本后的那次验证运行中，出现 **1 个错误**：

```
ERROR tests/test_login.py::test_locked_user_is_rejected[chromium]
playwright._impl._errors.TimeoutError: Page.goto: Timeout 30000ms exceeded.
  - navigating to "https://www.saucedemo.com/", waiting until "load"
======================== 10 passed, 1 error in 59.91s =========================
```

**与迁移和删除无关** —— 是被测站点（外部公开演示站 saucedemo.com）访问超时，代码根本没执行到断言。**复跑即恢复 `11 passed in 29.59s`**。

> 📌 **这暴露了一个真实的工程问题**：用例依赖外部站点时，网络抖动会被记成"测试失败"，污染测试结果的可信度。
>
> 两点建议：
> 1. 接入公司内网系统后，这个问题基本自然消失；
> 2. 若仍需依赖外部服务，可引入 `pytest-rerunfailures` 对网络类失败做有限次自动重试 —— 但要**谨慎区分"网络抖动"和"真实缺陷"**，无差别重试会掩盖真问题。
